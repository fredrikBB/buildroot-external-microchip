# BeagleV-Fire AMP Boot Environment
# Supports boot-time selection between base and AMP configurations

setenv fdt_high 0xffffffffffffffff
setenv initrd_high 0xffffffffffffffff

# Check for amp_mode environment variable
# Set with: setenv amp_mode 1; saveenv (for AMP mode)
# Or:       setenv amp_mode 0; saveenv (for base mode)

if test "${amp_mode}" = "1"; then
    echo "Booting AMP configuration...";
    load mmc 0:2 ${scriptaddr} mpfs_beaglev_fire_amp.itb;
    bootm start ${scriptaddr}#conf-amp;
else
    echo "Booting base configuration...";
    load mmc 0:2 ${scriptaddr} mpfs_beaglev_fire_amp.itb;
    bootm start ${scriptaddr}#conf-base;
fi;
bootm loados ${scriptaddr};
bootm ramdisk;
bootm prep;
bootm go;
